# 狗头军师 · 微信聊天副驾 (LSPosed 模块)

## 构建
1. 用 Android Studio (Koala 以上) 打开本目录，首次会自动下载 Gradle 并生成 wrapper
2. Build > Build APK(s)，或命令行 `./gradlew assembleDebug`
3. 安装 APK -> LSPosed「模块」里启用 -> 作用域勾选「微信」-> 强杀微信重开
4. 打开本 App：设置里填 API Key；首页点「作用域：微信」确认；到「试聊」页先验证接口通

## 工作方式
- hook com.tencent.mm 的 Activity.onResume（按类名含 LauncherUI/ChattingUI 识别聊天页，不依赖混淆类名）
- 每 1.5s 读取聊天列表：用头像位置判断对方/我（仅一对一聊天，群聊无头像规则会被跳过）
- 最后一条是对方消息时，调 LLM（狗头军师提示词）生成 3 条候选回复，显示在聊天页顶部卡片
- 点候选 = 填入输入框，**不会自动发送**；点卡片标题 = 重新生成；✕ = 本条不再显示

## 需要你按真机调的地方（Overlay.kt）
- 卡片顶部偏移 `dp(96)`：避开标题栏
- 头像尺寸判定 `dp(30)..dp(72)`：微信头像大小随字体/版本不同
- 微信更新后若读不到消息，先看 LSPosed 日志里 `[Goutou]` 开头的行

val DEFAULT_PROMPT = """
你是「狗头军师」，用户的微信聊天副驾。输入是最近的微信聊天记录（对方/我）。
流程：先判断对方这句话的真实意图和情绪；区分事实、推测、未知，不替对方脑补；评估回复风险；再给出可直接发送的回复。
规则：
- 回复要像真人发微信：口语、短句、1-2句，不要客服腔，不要“当然可以”，少用感叹号和表情。

- 信息不足时在 note 里说明，并让其中一条回复用来追问。
只输出 JSON，不要任何额外文字：
{"intent":"一句话","risk":"低/中/高","note":"一句话提醒","replies":[{"style":"稳妥","text":"..."},{"style":"幽默","text":"..."},{"style":"推进","text":"..."}]}
""".trimIndent()

class Config(ctx: Context) {

    }

package dev.goutou.wingman

import org.json.JSONArray
import org.json.JSONObject
import java.net.HttpURLConnection
import java.net.URL

data class Msg(val fromMe: Boolean, val text: String)
data class Reply(val style: String, val text: String)
data class Suggestion(val intent: String, val risk: String, val note: String, val replies: List<Reply>)

/** OpenAI 兼容接口（OpenAI / OpenRouter / DeepSeek / 自建中转都行）。 */
object Engine {
    fun analyze(c: ConfigData, msgs: List<Msg>): Suggestion {
        require(c.apiKey.isNotBlank()) { "未填写 API Key" }
        val chat = msgs.joinToString("\n") { (if (it.fromMe) "我: " else "对方: ") + it.text }
        val body = JSONObject()
            .put("model", c.model)
            .put("temperature", 0.8)
            .put(
                "messages", JSONArray()
                    .put(JSONObject().put("role", "system").put("content", c.prompt))
                    .put(JSONObject().put("role", "user").put("content", "聊天记录（时间顺序）：\n$chat\n\n请严格按系统要求只输出 JSON。"))
            )
        val out = post(endpoint(c.baseUrl), c.apiKey, body.toString())
        val content = JSONObject(out).getJSONArray("choices").getJSONObject(0)
            .getJSONObject("message").getString("content")
        return parse(content)
    }

    private fun endpoint(base: String): String {
        val u = base.trim().trimEnd('/')
        return if (u.endsWith("/chat/completions")) u else "$u/chat/completions"
    }

    private fun post(url: String, key: String, json: String): String {
        val conn = URL(url).openConnection() as HttpURLConnection
        conn.requestMethod = "POST"
        conn.connectTimeout = 15000
        conn.readTimeout = 40000
        conn.doOutput = true
        conn.setRequestProperty("Content-Type", "application/json")
        conn.setRequestProperty("Authorization", "Bearer $key")
        conn.outputStream.use { it.write(json.toByteArray()) }
        val code = conn.responseCode
        val stream = if (code in 200..299) conn.inputStream else conn.errorStream
        val text = stream?.bufferedReader()?.use { it.readText() } ?: ""
        if (code !in 200..299) error("HTTP $code ${text.take(160)}")
        return text
    }

    private fun parse(raw: String): Suggestion {
        val s = raw.indexOf('{')
        val e = raw.lastIndexOf('}')
        require(s >= 0 && e > s) { "模型没有返回 JSON" }
        val o = JSONObject(raw.substring(s, e + 1))
        val arr = o.optJSONArray("replies") ?: JSONArray()
        val replies = (0 until arr.length()).map {
            val r = arr.getJSONObject(it)
            Reply(r.optString("style"), r.optString("text"))
        }.filter { it.text.isNotBlank() }
        return Suggestion(o.optString("intent"), o.optString("risk"), o.optString("note"), replies)
    }
}

object Overlay {
    private val map = WeakHashMap<Activity, Panel>()
    fun attach(a: Activity) {
        if (map.containsKey(a)) return
        val p = Panel(a)
        map[a] = p
        p.install()
    }
}

/** 悬浮在聊天页顶部的候选回复卡片。只读消息、只填输入框，不自动发送。 */
class Panel(private val a: Activity) {
    private val h = Handler(Looper.getMainLooper())
    private val card = LinearLayout(a)
    private val title = TextView(a)
    private val body = LinearLayout(a)
    private val prefs = XSharedPreferences(MODULE_PKG, PREF_NAME)
    private var lastKey = ""
    private var dismissedKey = ""
    @Volatile private var busy = false
    private val d = a.resources.displayMetrics.density
    private fun dp(x: Int) = (x * d).toInt()

    private val tick = object : Runnable {
        override fun run() {
            if (a.isFinishing || a.isDestroyed) return
            try { scan() } catch (t: Throwable) { XposedBridge.log(t) }
            h.postDelayed(this, 1500)
        }
    }

    fun install() {
        card.orientation = LinearLayout.VERTICAL
        card.background = GradientDrawable().apply {
            cornerRadius = dp(22).toFloat(); setColor(0xF5FFFFFF.toInt()); setStroke(dp(1), 0x667C3AED)
        }
        card.setPadding(dp(14), dp(10), dp(14), dp(10))
        card.elevation = dp(8).toFloat()
        card.visibility = View.GONE

        val head = LinearLayout(a).apply { orientation = LinearLayout.HORIZONTAL; gravity = Gravity.CENTER_VERTICAL }
        title.setTextColor(0xFF7C3AED.toInt()); title.textSize = 13f
        title.setOnClickListener { lastKey = ""; dismissedKey = "" } // 点标题重新生成
        val close = TextView(a).apply { text = "✕"; textSize = 14f; setTextColor(0xFF888888.toInt()); setPadding(dp(12), 0, 0, 0) }
        close.setOnClickListener { dismissedKey = lastKey; card.visibility = View.GONE }
        head.addView(title, LinearLayout.LayoutParams(0, ViewGroup.LayoutParams.WRAP_CONTENT, 1f))
        head.addView(close)
        body.orientation = LinearLayout.VERTICAL
        card.addView(head)
        card.addView(body)

        val lp = FrameLayout.LayoutParams(ViewGroup.LayoutParams.MATCH_PARENT, ViewGroup.LayoutParams.WRAP_CONTENT, Gravity.TOP)
        lp.setMargins(dp(10), dp(96), dp(10), 0) // 顶部偏移：避开状态栏+标题栏，按机型调
        (a.window.decorView as ViewGroup).addView(card, lp)
        h.postDelayed(tick, 1500)
    }

    private fun scan() {
        val decor = a.window.decorView
        if (!decor.hasWindowFocus()) return
        if (!prefs.file.canRead()) { show("配置读取失败：请确认模块已在 LSPosed 启用且勾选了微信"); return }
        prefs.reload()
        val cfg = ConfigData.from(prefs)
        if (!cfg.enabled || findEdit(decor) == null) { card.visibility = View.GONE; return }

        val list = findList(decor) ?: run { card.visibility = View.GONE; return }
        val msgs = readRows(list).takeLast(cfg.ctx)
        if (msgs.isEmpty() || msgs.last().fromMe) { card.visibility = View.GONE; return }

        val key = msgs.joinToString("|") { (if (it.fromMe) "M" else "O") + it.text }
        if (key == dismissedKey) return
        if (key == lastKey || busy) return
        lastKey = key
        busy = true
        show("狗头军师思考中…")
        Thread {
            try {
                val s = Engine.analyze(cfg, msgs)
                h.post { render(s) }
            } catch (t: Throwable) {
                h.post { show("生成失败：${t.message}（点这里重试）") }
            } finally { busy = false }
        }.start()
    }

    private fun show(t: String) {
        title.text = t; body.removeAllViews(); card.visibility = View.VISIBLE
    }

    private fun render(s: Suggestion) {
        title.text = "狗头军师 · ${s.intent} · 风险${s.risk}"
        body.removeAllViews()
        if (s.note.isNotBlank()) body.addView(TextView(a).apply {
            text = s.note; textSize = 12f; setTextColor(0xFF666666.toInt()); setPadding(0, dp(4), 0, dp(2))
        })
        for (r in s.replies.take(3)) {
            val tv = TextView(a).apply {
                text = "${r.style}｜${r.text}"; textSize = 14f; setTextColor(0xFF222222.toInt())
                setPadding(dp(12), dp(8), dp(12), dp(8))
                background = GradientDrawable().apply { cornerRadius = dp(14).toFloat(); setColor(0xFFEDE7FA.toInt()) }
                setOnClickListener { fill(r.text) }
            }
            val lp = LinearLayout.LayoutParams(ViewGroup.LayoutParams.MATCH_PARENT, ViewGroup.LayoutParams.WRAP_CONTENT)
            lp.topMargin = dp(6)
            body.addView(tv, lp)
        }
        card.visibility = View.VISIBLE
    }

    private fun fill(text: String) {
        val et = findEdit(a.window.decorView) ?: return
        et.setText(text); et.setSelection(text.length); et.requestFocus()
    }

    // ---------- 读取聊天 ----------
    private fun findEdit(v: View): EditText? {
        if (v is EditText && v.isShown) return v
        if (v is ViewGroup) for (i in 0 until v.childCount) findEdit(v.getChildAt(i))?.let { return it }
        return null
    }

    private fun findList(v: View): ViewGroup? {
        if (v is ViewGroup) {
            if ((v is AbsListView || v.javaClass.name.contains("RecyclerView")) && v.isShown && v.childCount > 1) return v
            for (i in 0 until v.childCount) findList(v.getChildAt(i))?.let { return it }
        }
        return null
    }

    private val noise = Regex("""^\d+\s*["'″′”“]*$""")
    private val timeRe = Regex("""^((昨天|今天|星期.|周.|\d{4}年)?\s*\d{1,2}[:：]\d{2}|\d{1,2}月\d{1,2}日.*)$""")

    /** 按头像位置判方向：头像在屏幕左半=对方，右半=我。群聊/无头像的行直接跳过。 */
    private fun readRows(list: ViewGroup): List<Msg> {
        val w = a.resources.displayMetrics.widthPixels
        val out = ArrayList<Msg>()
        for (i in 0 until list.childCount) {
            val row = list.getChildAt(i)
            var avatarX = -1
            var best = ""
            fun walk(v: View) {
                if (!v.isShown) return
                if (v is ImageView && avatarX < 0) {
                    val sz = v.width
                    if (sz in dp(30)..dp(72) && Math.abs(v.width - v.height) < dp(4)) {
                        val loc = IntArray(2); v.getLocationOnScreen(loc); avatarX = loc[0] + sz / 2
                    }
                }
                if (v is TextView && v !is EditText) {
                    val t = v.text?.toString()?.trim().orEmpty()
                    if (t.length > best.length) best = t
                }
                if (v is ViewGroup) for (k in 0 until v.childCount) walk(v.getChildAt(k))
            }
            walk(row)
            if (avatarX < 0 || best.isEmpty() || best.length > 300) continue
            if (timeRe.matches(best) || noise.matches(best)) continue
            out.add(Msg(fromMe = avatarX > w / 2, text = best))
        }
        return out
    }
}

@Composable
fun HomeScreen(cfg: Config, onUltra: () -> Unit) {
    val ctx = LocalContext.current
    var d by remember { mutableStateOf(cfg.load()) }
    var filter by remember { mutableIntStateOf(0) } // 0 全部 1 危险 2 可疑 3 完美
    val active = ModuleStatus.isActive()

    val checks = listOf(
        Check("模块激活", if (active) "LSPosed 已加载本模块" else "未激活：请在 LSPosed 启用模块后重启本应用", if (active) Level.OK else Level.BAD),
        Check(
            "作用域：微信",
            if (d.scopeOk) "已确认勾选 com.tencent.mm（点击取消确认）" else "请在 LSPosed 勾选微信并重启微信，然后点击此处确认",
            if (d.scopeOk) Level.OK else Level.WARN
        ) { cfg.save(d.copy(scopeOk = !d.scopeOk)); d = cfg.load() },
        Check("API Key", if (d.apiKey.isBlank()) "未填写，请到「设置」填写" else "已填写（仅存本机）", if (d.apiKey.isBlank()) Level.BAD else Level.OK),
        Check("接口与模型", "${d.model} · ${d.baseUrl}", if (d.baseUrl.startsWith("http")) Level.OK else Level.BAD),
        Check(
            "军师提示词",
            "${d.prompt.length} 字 · " + if (d.prompt == DEFAULT_PROMPT) "内置精简版" else "自定义",
            if (d.prompt.length < 50) Level.WARN else Level.OK
        ),
        Check("自动分析", if (d.enabled) "进入微信聊天页自动生成候选回复" else "已关闭", if (d.enabled) Level.OK else Level.WARN),
    )
    val bad = checks.count { it.level == Level.BAD }
    val warn = checks.count { it.level == Level.WARN }
    val ok = checks.count { it.level == Level.OK }
    val overall = if (bad > 0) Level.BAD else if (warn > 0) Level.WARN else Level.OK
    val shown = when (filter) { 1 -> checks.filter { it.level == Level.BAD }; 2 -> checks.filter { it.level == Level.WARN }; 3 -> checks.filter { it.level == Level.OK }; else -> checks }

    LazyColumn(contentPadding = PaddingValues(bottom = 130.dp)) {
        item {
            Header("狗头军师", "WeChat · 聊天副驾") {
                IconButton({ d = cfg.load() }, Modifier.padding(top = 8.dp).size(52.dp).clip(CircleShape).background(PurpleSoft)) {
                    Icon(Icons.Filled.Refresh, null, tint = TextMain)
                }
                Spacer(Modifier.width(12.dp))
                Button(onUltra, Modifier.padding(top = 4.dp).height(60.dp), shape = RoundedCornerShape(24.dp), colors = ButtonDefaults.buttonColors(containerColor = Purple)) {
                    Icon(Icons.Filled.AutoAwesome, null); Spacer(Modifier.width(6.dp)); Text("试聊", fontSize = 18.sp, fontWeight = FontWeight.Bold)
                }
            }
        }
        item {
            Card(border = overall.color.copy(alpha = .45f)) {
                Row(verticalAlignment = Alignment.CenterVertically) {
                    Box(Modifier.size(84.dp).clip(RoundedCornerShape(26.dp)).background(overall.color), contentAlignment = Alignment.Center) {
                        Icon(if (overall == Level.OK) Icons.Filled.Check else Icons.Filled.Warning, null, tint = Color.White, modifier = Modifier.size(48.dp))
                    }
                    Spacer(Modifier.width(18.dp))
                    Column(Modifier.weight(1f)) {
                        Text(when (overall) { Level.OK -> "就绪!"; Level.WARN -> "待确认"; else -> "有问题" }, fontSize = 36.sp, fontWeight = FontWeight.Bold, color = overall.color)
                        Text(when (overall) { Level.OK -> "ALL CLEAR"; Level.WARN -> "NEEDS CHECK"; else -> "ACTION NEEDED" }, fontSize = 16.sp, letterSpacing = 3.sp, color = overall.color, fontWeight = FontWeight.Medium)
                        Text("${checks.size} 项已检测", fontSize = 16.sp, color = TextSub)
                    }
                }
                Spacer(Modifier.height(14.dp))
                Text(
                    when (overall) { Level.OK -> "配置完整，微信聊天页将自动出现候选回复。"; Level.WARN -> "部分项目需要你确认，点下方条目处理。"; else -> "有必需项未通过，模块暂时无法工作。" },
                    fontSize = 17.sp, color = TextMain
                )
                Spacer(Modifier.height(14.dp))
                Row(horizontalArrangement = Arrangement.spacedBy(10.dp)) {
                    Stat("${checks.size} 项", "已检测", Modifier.weight(1f))
                    Stat("$ok 项", "完美", Modifier.weight(1f))
                    Stat("${bad + warn} 项", "待处理", Modifier.weight(1f))
                }
            }
        }
        item {
            Card(Modifier.clickable { ctx.startActivity(Intent(Intent.ACTION_VIEW, Uri.parse("https://github.com/shengjidaguai-china/goutoujunshi"))) }) {
                Row(verticalAlignment = Alignment.CenterVertically) {
                    Icon(Icons.Filled.HelpOutline, null, tint = GreenOk, modifier = Modifier.size(40.dp))
                    Spacer(Modifier.width(14.dp))
                    Column(Modifier.weight(1f)) {
                        Text("需要帮助？查看狗头军师项目", fontSize = 18.sp, fontWeight = FontWeight.Medium)
                        Text("点击跳转 GitHub 获取完整知识库", fontSize = 15.sp, color = TextSub)
                    }
                    Icon(Icons.Filled.OpenInNew, null, tint = TextSub)
                }
            }
        }
        item {
            Card {
                Row(Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.SpaceBetween) {
                    listOf("全部 ${checks.size}", "危险 $bad", "可疑 $warn", "完美 $ok").forEachIndexed { i, t ->
                        Pill(t, filter == i) { filter = i }
                    }
                }
            }
        }
        item { Text("检测项目", Modifier.padding(start = 28.dp, top = 12.dp, bottom = 4.dp), fontSize = 20.sp, color = Purple, fontWeight = FontWeight.Medium) }
        items(shown) { c -> CheckRow(c) }
    }
}

@Composable
fun Stat(big: String, small: String, modifier: Modifier) {
    Column(modifier.clip(RoundedCornerShape(24.dp)).background(PurpleSoft).padding(vertical = 14.dp), horizontalAlignment = Alignment.CenterHorizontally) {
        Text(big, fontSize = 24.sp, fontWeight = FontWeight.Medium)
        Text(small, fontSize = 14.sp, color = TextSub)
    }
}

@Composable
fun CheckRow(c: Check) {
    Surface(
        Modifier.fillMaxWidth().padding(horizontal = 16.dp, vertical = 5.dp).clip(RoundedCornerShape(28.dp)).clickable(enabled = c.onClick != null) { c.onClick?.invoke() },
        shape = RoundedCornerShape(28.dp), color = CardBg
    ) {
        Row(Modifier.padding(16.dp), verticalAlignment = Alignment.CenterVertically) {
            Box(Modifier.size(64.dp).clip(RoundedCornerShape(20.dp)).background(c.level.color.copy(alpha = .15f)), contentAlignment = Alignment.Center) {
                Box(Modifier.size(40.dp).clip(CircleShape).background(c.level.color), contentAlignment = Alignment.Center) {
                    Icon(if (c.level == Level.OK) Icons.Filled.Check else Icons.Filled.PriorityHigh, null, tint = Color.White)
                }
            }
            Spacer(Modifier.width(16.dp))
            Column(Modifier.weight(1f)) {
                Text(c.title, fontSize = 20.sp, fontWeight = FontWeight.Bold)
                Text(c.desc, fontSize = 15.sp, color = TextSub, maxLines = 1, overflow = TextOverflow.Ellipsis)
            }
            Text(if (c.level == Level.OK) "通过" else c.level.label, color = c.level.color, fontSize = 16.sp, fontWeight = FontWeight.Medium)
        }
    }
}
